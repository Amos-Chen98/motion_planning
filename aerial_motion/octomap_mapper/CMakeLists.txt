cmake_minimum_required(VERSION 3.10)
project(octomap_mapper)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS roscpp std_srvs octomap_server)
find_package(octomap REQUIRED CONFIG)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS})
add_executable(octomap_server_adapter src/octomap_server_adapter.cpp)
target_link_libraries(octomap_server_adapter ${catkin_LIBRARIES} octomap)
add_dependencies(octomap_server_adapter ${catkin_EXPORTED_TARGETS})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  add_rostest(test/mapper_node.test)
  catkin_install_python(PROGRAMS test/mapper_node_test.py
    DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
endif()
install(TARGETS octomap_server_adapter RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
